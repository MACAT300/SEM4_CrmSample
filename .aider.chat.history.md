package com.example.crm.customer.repository

import com.example.crm.customer.model.Customer
import java.util.concurrent.ConcurrentHashMap

class InMemoryCustomerRepository : CustomerRepository {
    private val customers = ConcurrentHashMap<String, Customer>()

        customers[customer.id] = customer
        return customer
    }

    override fun update(customer: Customer): Customer {

        return customer
    }

    override fun findById(id: String): Customer? {

    }

    override fun findAll(): List<Customer> {

}

package com.example.crm.lead.service

import com.example.crm.lead.model.Lead
import com.example.crm.lead.repository.LeadRepository
import com.example.crm.customer.model.Customer
import com.example.crm.customer.repository.CustomerRepository
import com.example.crm.customer.service.CustomerService
import java.time.LocalDateTime

class LeadServiceImpl(
    private val leadRepository: LeadRepository,
    private val customerRepository: CustomerRepository,
    private val customerService: CustomerService
) : LeadService {
    override fun createLead(lead: Lead): Lead {
        validateLead(lead)
        return leadRepository.create(lead)
    }

    override fun assignLead(leadId: String, salespersonId: String): Lead {
        val lead = leadRepository.findById(leadId) ?: throw IllegalArgumentException("Lead not found")
        val updatedLead = lead.copy(status = LeadStatus.ASSIGNED, assignedSalesperson = salespersonId)
        return leadRepository.update(updatedLead)
    }

    override fun convertLeadToCustomer(leadId: String): Customer {
        val lead = leadRepository.findById(leadId) ?: throw IllegalArgumentException("Lead not found")
        val customer = Customer(
            id = lead.id,
            name = lead.name,
            phone = lead.contactInfo,
            email = lead.contactInfo,
            status = CustomerStatus.ACTIVE,
            createdAt = LocalDateTime.now()
        )
        return customerService.createCustomer(customer)
    }

    override fun listLeads(): List<Lead> {
        return leadRepository.findAll()
    }

    private fun validateLead(lead: Lead) {
        require(lead.id.isNotBlank()) { "Lead ID cannot be blank" }
        require(lead.name.isNotBlank()) { "Lead name cannot be blank" }
        require(lead.contactInfo.isNotBlank()) { "Lead contact info cannot be blank" }
        require(lead.source.isNotBlank()) { "Lead source cannot be blank" }
        require(lead.status != null) { "Lead status cannot be null" }
        require(lead.createdAt != null) { "Lead creation date cannot be null" }
    }
}

package com.example.crm.lead.service

import com.example.crm.lead.model.Lead
import com.example.crm.lead.repository.LeadRepository
import com.example.crm.customer.model.Customer
import com.example.crm.customer.repository.CustomerRepository
import com.example.crm.customer.service.CustomerService
import java.time.LocalDateTime

class LeadServiceImpl(
    private val leadRepository: LeadRepository,
    private val customerRepository: CustomerRepository,
    private val customerService: CustomerService
) : LeadService {
    override fun createLead(lead: Lead): Lead {
        validateLead(lead)
        return leadRepository.create(lead)
    }

    override fun assignLead(leadId: String, salespersonId: String): Lead {
        val lead = leadRepository.findById(leadId) ?: throw IllegalArgumentException("Lead not found")
        val updatedLead = lead.copy(status = LeadStatus.ASSIGNED, assignedSalesperson = salespersonId)
        return leadRepository.update(updatedLead)
    }

    override fun convertLeadToCustomer(leadId: String): Customer {
        val lead = leadRepository.findById(leadId) ?: throw IllegalArgumentException("Lead not found")
        val customer = Customer(
            id = lead.id,
            name = lead.name,
            phone = lead.contactInfo,
            email = lead.contactInfo,
            status = CustomerStatus.ACTIVE,
            createdAt = LocalDateTime.now()
        )
        return customerService.createCustomer(customer)
    }

    override fun listLeads(): List<Lead> {
        return leadRepository.findAll()
    }

    private fun validateLead(lead: Lead) {
        require(lead.id.isNotBlank()) { "Lead ID cannot be blank" }
        require(lead.name.isNotBlank()) { "Lead name cannot be blank" }
        require(lead.contactInfo.isNotBlank()) { "Lead contact info cannot be blank" }

import com.example.crm.activity.model.Activity
import com.example.crm.activity.repository.InMemoryActivityRepository
import com.example.crm.activity.service.ActivityServiceImpl
import com.example.crm.activity.usecase.CreateActivityUseCase
import com.example.crm.activity.usecase.ListActivitiesUseCase
import org.junit.jupiter.api.Assertions.*
import org.junit.jupiter.api.BeforeEach
import org.junit.jupiter.api.Test
import java.time.LocalDateTime

class ActivityUseCaseTest {
    private lateinit var activityRepository: ActivityRepository
    private lateinit var activityService: ActivityService
    private lateinit var createActivityUseCase: CreateActivityUseCase
    private lateinit var listActivitiesUseCase: ListActivitiesUseCase

    @BeforeEach
    fun setUp() {
        activityRepository = InMemoryActivityRepository()
        activityService = ActivityServiceImpl(activityRepository)
        createActivityUseCase = CreateActivityUseCase(activityService)
        listActivitiesUseCase = ListActivitiesUseCase(activityService)
    }

    @Test
    fun `test create activity with valid data`() {
        val activity = Activity(
            id = "1",
            customerId = "1",
            description = "Meeting with John Doe",
            createdAt = LocalDateTime.now()
        )
        val createdActivity = createActivityUseCase.execute(activity)
        assertEquals(activity, createdActivity)
    }

    @Test
    fun `test create activity with invalid data`() {
        val activity = Activity(
            id = "",
            customerId = "1",
            description = "Meeting with John Doe",
            createdAt = LocalDateTime.now()
        )
        assertThrows(IllegalArgumentException::class.java) {
            createActivityUseCase.execute(activity)
        }
    }

    @Test
    fun `test list all activities`() {
        val activity1 = Activity(
            id = "1",
            customerId = "1",
            description = "Meeting with John Doe",
            createdAt = LocalDateTime.now()
        )
        val activity2 = Activity(
            id = "2",
            customerId = "2",
            description = "Follow-up with Jane Doe",
            createdAt = LocalDateTime.now()
        )
        createActivityUseCase.execute(activity1)
        createActivityUseCase.execute(activity2)
        val activities = listActivitiesUseCase.execute()
        assertEquals(2, activities.size)
        assertTrue(activities.contains(activity1))
        assertTrue(activities.contains(activity2))
    }
}
